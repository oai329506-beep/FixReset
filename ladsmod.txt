local LADSModPopup = {}

local FORM_PATH = "UGUI/Form/System/ILua/ILuaUniversal/Form_MessageBox.prefab"
local YOUTUBE_URL = "https://youtube.com/@motmod2008?si=nJhv72Wdxal4MoEZ"

local MOD_RESOURCE_PATHS = rawget(_G, "LADS_MOD_RESOURCE_PATHS") or {
}

local MOD_DLC_TYPES = rawget(_G, "LADS_MOD_DLC_TYPES") or {
}

local TITLE = "<color=#FFD700>Hạn sử dụng: Còn</color>"
local CONTENT = table.concat({
  "File mod được hoàn thiện bởi <color=#00FF00><i><b>LADS MOD</b></i></color>",
  "Ấn <color=#FFFF00><i>Đã Hiểu</i></color> để tiếp tục sử dụng mod.",
  "Ấn <color=#FF0000><i>Kênh Youtube</i></color> để truy cập kênh.",
  "Ấn <color=#FF4040><i>Xoá Mod</i></color> để xoá tài nguyên mod và tải lại tài nguyên gốc."
}, "\n")

local function closePopup()
  N.OLDSYS_CUIManager.instance:CloseForm(FORM_PATH)
end

local function isPathInsideRoot(root, fullPath)
  root = tostring(CS.System.IO.Path.GetFullPath(root))
  fullPath = tostring(CS.System.IO.Path.GetFullPath(fullPath))
  local sep = tostring(CS.System.IO.Path.DirectorySeparatorChar)
  if fullPath == root then
    return false
  end
  if string.sub(root, -1) ~= sep then
    root = root .. sep
  end
  return string.sub(fullPath, 1, #root) == root
end

local function deleteConfiguredResources()
  if #MOD_RESOURCE_PATHS == 0 then
    N.OLDSYS_CUIManager.instance:OpenTips("Chưa cấu hình đường dẫn tài nguyên mod cần xoá.")
    return false
  end

  local root = tostring(CS.UnityEngine.Application.persistentDataPath)
  local removed = 0
  local failed = 0

  for _, relativePath in ipairs(MOD_RESOURCE_PATHS) do
    local ok, err = pcall(function()
      local fullPath = tostring(CS.System.IO.Path.GetFullPath(CS.System.IO.Path.Combine(root, relativePath)))
      if not isPathInsideRoot(root, fullPath) then
        error("Đường dẫn nằm ngoài persistentDataPath: " .. tostring(relativePath))
      end

      if CS.System.IO.File.Exists(fullPath) then
        CS.System.IO.File.Delete(fullPath)
        removed = removed + 1
      elseif CS.System.IO.Directory.Exists(fullPath) then
        CS.System.IO.Directory.Delete(fullPath, true)
        removed = removed + 1
      end
    end)

    if not ok then
      failed = failed + 1
      print_err("[LADSModPopup] delete failed: {0}", tostring(err))
    end
  end

  for _, dlcType in ipairs(MOD_DLC_TYPES) do
    pcall(function()
      N.DLCDownloadUISystem.instance:TryDownloadDLCByType(dlcType)
    end)
  end

  if removed > 0 then
    N.OLDSYS_CUIManager.instance:OpenTips(
      string.format("Đã xoá %d mục tài nguyên mod. Hãy khởi động lại game để tải lại tài nguyên gốc.", removed)
    )
    return true
  end

  if failed > 0 then
    N.OLDSYS_CUIManager.instance:OpenTips("Không thể xoá một số tài nguyên mod.")
  else
    N.OLDSYS_CUIManager.instance:OpenTips("Không tìm thấy tài nguyên mod cần xoá.")
  end
  return false
end

local function askDeleteMod()
  N.OLDSYS_CUIManager.GetInstance():OpenMessageBoxCallBack(
    "<color=#FF4040><b>Xoá Mod?</b></color>\nTài nguyên mod đã cấu hình sẽ bị xoá. Game có thể cần tải lại tài nguyên sau khi khởi động lại.",
    true,
    function()
      local deleted = deleteConfiguredResources()
      if deleted then
        closePopup()
      end
    end,
    nil,
    "<color=#FF4040>Xoá Mod</color>",
    "Huỷ"
  )
end

local function getRect(go)
  if not go then
    return nil
  end
  return go:GetComponent(typeof(CS.UnityEngine.RectTransform))
end

local function setButtonText(go, text)
  if not go then
    return
  end
  local label = go:GetComponentInChildren(typeof(CS.Text2))
  if label then
    label.text = text
  end
end

local function arrangeThreeButtons(cancel, confirm, deleteBtn)
  local cancelRT = getRect(cancel)
  local confirmRT = getRect(confirm)
  local deleteRT = getRect(deleteBtn)
  if not cancelRT or not confirmRT or not deleteRT then
    return
  end

  local y = confirmRT.anchoredPosition.y
  local spacing = 285
  cancelRT.anchoredPosition = CS.UnityEngine.Vector2(-spacing, y)
  confirmRT.anchoredPosition = CS.UnityEngine.Vector2(0, y)
  deleteRT.anchoredPosition = CS.UnityEngine.Vector2(spacing, y)
end

local function enlargeContent(content)
  local rt = getRect(content)
  if not rt then
    return
  end
  local size = rt.sizeDelta
  rt.sizeDelta = CS.UnityEngine.Vector2(size.x + 120, size.y + 90)
end

function LADSModPopup.Open()
  local form = N.OLDSYS_CUIManager.instance:OpenForm(FORM_PATH, false, false)
  if not form then
    return nil
  end

  local title = form:GetWidget(0)
  local content = form:GetWidget(1)
  local toggle = form:GetWidget(2)
  local close = form:GetWidget(3)
  local cancel = form:GetWidget(4)
  local confirm = form:GetWidget(5)
  local icon = form:GetWidget(6)

  if title then
    title:GetComponent(typeof(CS.Text2)).text = TITLE
  end

  if content then
    content:GetComponent(typeof(CS.Text2)).text = CONTENT
    enlargeContent(content)
  end

  if toggle then
    toggle:CustomSetActive(false)
  end

  if icon then
    icon:CustomSetActive(false)
  end

  if close then
    close:CustomSetActive(true)
  end

  cancel:CustomSetActive(true)
  confirm:CustomSetActive(true)

  setButtonText(cancel, "<color=#FFFF00>Đã Hiểu</color>")
  setButtonText(confirm, "<color=#FF0000>Kênh Youtube</color>")

  local cancelEvent = cancel:GetComponent(typeof(N.OLDSYS_CUIEventScript))
  local confirmEvent = confirm:GetComponent(typeof(N.OLDSYS_CUIEventScript))

  if cancelEvent then
    cancelEvent.onClick = function()
      closePopup()
    end
  end

  if confirmEvent then
    confirmEvent.onClick = function()
      CS.UnityEngine.Application.OpenURL(YOUTUBE_URL)
    end
  end

  local deleteBtn = CS.UnityEngine.Object.Instantiate(cancel, cancel.transform.parent)
  deleteBtn.name = "BtnDeleteMod"
  deleteBtn:CustomSetActive(true)
  setButtonText(deleteBtn, "<color=#FF4040>Xoá Mod</color>")

  local deleteEvent = deleteBtn:GetComponent(typeof(N.OLDSYS_CUIEventScript))
  if deleteEvent then
    deleteEvent.onClick = function()
      askDeleteMod()
    end
  end

  arrangeThreeButtons(cancel, confirm, deleteBtn)
  return form
end

return LADSModPopup
